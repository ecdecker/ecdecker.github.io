{{- /* Token gallery. Every swatch is read out of assets/css/tokens.css at build time rather than transcribed here. A hand-written gallery drifts the moment someone adds a token and forgets to update it, and a drifted gallery is worse than none — it reports coverage it does not have. This one cannot drift: if a custom property exists in :root it appears below, and if it is deleted it disappears. Values are never printed from the parse. Each specimen renders var(--name) and lets the browser resolve it, so what is shown is the value actually in effect, not a copy of it. The raw declaration is printed alongside only as a label. */ -}} {{ define "main" }} {{- $css := (resources.Get "css/tokens.css").Content -}} {{- /* Only the :root block. The html.type-system override at the bottom of the file redefines --font-display and would produce a duplicate. */ -}} {{- $root := $css -}} {{- with findRESubmatch `:root\s*\{([^}]*)\}` $css 1 -}} {{- $root = index (index . 0) 1 -}} {{- end -}} {{- $tokens := slice -}} {{- range findRESubmatch `--([a-z0-9-]+)\s*:\s*([^;]+);` $root -}} {{- $tokens = $tokens | append (dict "name" (index . 1) "value" (trim (index . 2) " \n\t")) -}} {{- end -}}
