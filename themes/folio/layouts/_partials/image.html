{{- /* Optimized responsive image. Emits AVIF with a WebP fallback and no original-format source. The original-format that this pattern usually carries adds no capability — a browser that skips the modern sources matches the fallback, which is already that format — while costing the most bytes of anything on the page. Params: resource (resource) an already-resolved image, e.g. a page-bundle sibling. Takes precedence over src. src (string) path under assets/, with or without leading / alt (string, required) empty string for decorative images sizes (string) media-condition sizes attribute. Defaults to 100vw, which over-fetches — always pass a real value. class (string) classes for the widths (slice) candidate widths; capped to the source width quality (int) encoder quality; overrides the measured per-image value from data/imagequality.json eager (bool) true for the one above-the-fold image fit (string) "resize" (default) or "fill" ratio (string) target aspect ratio for fit=fill, e.g. "16:9" Example: {{ partial "image.html" (dict "src" "images/plate.png" "alt" "Field site" "sizes" "(min-width: 880px) 640px, 100vw") }} */ -}} {{- $cfg := site.Params.images | default dict -}} {{- $src := strings.TrimPrefix "/" .src -}} {{- $alt := .alt | default "" -}} {{- $class := .class | default "" -}} {{- $sizes := .sizes | default ($cfg.sizes | default "100vw") -}} {{- $quality := .quality | default ($cfg.quality | default 45) -}} {{- $fit := .fit | default "resize" -}} {{- $ratio := .ratio | default "" -}} {{- $eager := .eager | default false -}} {{- $widths := .widths | default ($cfg.widths | default (slice 320 480 640 800 1200)) -}} {{- $resolved := .resource -}} {{- if not $resolved }}{{ $resolved = resources.Get $src }}{{ end -}} {{- with $resolved }} {{- $img := . -}} {{- /* Per-image quality. One global q is a bad trade: holding a fixed perceptual floor (SSIM 0.92) takes q25 on a soft gradient and q70 on a fine texture, so a single value either wastes bytes on the easy images or wrecks the hard ones. Measured per image by tools/measure-image-quality.py --write-data; on this corpus that is 1.70x fewer bytes than the cheapest global q clearing the same floor. The key is the source path relative to assets/ or content/. A global asset's .Name is already "/images/plate.png"; a page-bundle sibling's .Name is the bare "plate.png", so it needs the owning page's directory. Both were verified against Hugo 0.164 — do not swap in .RelPermalink, which carries the language prefix and any slug, neither of which the source-path key has. An explicit quality param still wins, and anything unmeasured falls back to the global default. */ -}} {{- if not $.quality }} {{- $key := "" -}} {{- if hasPrefix $img.Name "/" -}} {{- $key = $img.Name -}} {{- else -}} {{- with page.File }}{{ $key = path.Join .Dir $img.Name }}{{ end -}} {{- end -}} {{- $key = strings.TrimPrefix "/" $key -}} {{- $table := dict -}} {{- /* hugo.Data, not site.Data: the latter is deprecated in 0.156+ and builds must stay warning-free. */ -}} {{- with hugo.Data.imagequality }}{{ with .images }}{{ $table = . }}{{ end }}{{ end -}} {{- with and $key (index $table $key) }} {{- if reflect.IsMap . -}} {{- with .quality }}{{ $quality = int . }}{{ end -}} {{- else -}} {{- $quality = int . -}} {{- end -}} {{- end }} {{- end }} {{- /* Never upscale: a width above the source invents pixels and costs real bytes. Always keep at least one candidate. */ -}} {{- $usable := slice -}} {{- range $widths }} {{- if le . $img.Width }}{{ $usable = $usable | append . }}{{ end }} {{- end }} {{- if not $usable }}{{ $usable = slice $img.Width }}{{ end }} {{- /* fit=fill needs a ratio; without one it silently crops to nothing useful, so fall back to a plain resize and say so. */ -}} {{- if and (eq $fit "fill") (not $ratio) }} {{- warnf "image.html: fit=fill needs a ratio for %s — resizing instead." $src }} {{- $fit = "resize" }} {{- end }} {{- $rw := 0 }}{{- $rh := 0 }} {{- if eq $fit "fill" }} {{- $parts := split $ratio ":" -}} {{- $rw = index $parts 0 | int -}} {{- $rh = index $parts 1 | int -}} {{- end }} {{- $avif := slice }}{{- $webp := slice }}{{- $largest := "" -}} {{- range $w := $usable }} {{- $spec := "" -}} {{- if eq $fit "fill" -}} {{- $spec = printf "%dx%d Center" $w (div (mul $w $rh) $rw) -}} {{- else -}} {{- $spec = printf "%dx" $w -}} {{- end -}} {{- $a := "" }}{{- $p := "" -}} {{- if eq $fit "fill" -}} {{- $a = $img.Fill (printf "%s avif q%d" $spec $quality) -}} {{- $p = $img.Fill (printf "%s webp q%d" $spec $quality) -}} {{- else -}} {{- $a = $img.Resize (printf "%s avif q%d" $spec $quality) -}} {{- $p = $img.Resize (printf "%s webp q%d" $spec $quality) -}} {{- end -}} {{- $avif = $avif | append (printf "%s %dw" $a.RelPermalink $w) -}} {{- $webp = $webp | append (printf "%s %dw" $p.RelPermalink $w) -}} {{- $largest = $p -}} {{- end }} {{- else }} {{- warnf "image.html: no asset at %q — image omitted." $src }} {{- end }}
